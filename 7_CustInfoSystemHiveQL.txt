CREATE TABLE customer_info (
    cust_id INT,
    cust_name STRING,
    order_id INT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE;


-- =====================================
-- Create order_info Table
-- =====================================

CREATE TABLE order_info (
    order_id INT,
    item_id INT,
    quantity INT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ',';

-- =====================================
-- Create item_info Table
-- =====================================

CREATE TABLE item_info (
    item_id INT,
    item_name STRING,
    item_price INT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ',';

-- =====================================

-- =====================================
-- View Table Data
-- =====================================

select * from customer_info;

select * from order_info;

select * from item_info;

-- =====================================
-- Join Tables
-- =====================================

SELECT
c.cust_id,
c.cust_name,
o.order_id,
i.item_name,
o.quantity,
i.item_price,
(o.quantity * i.item_price) AS total_cost

FROM customer_info c

JOIN order_info o
ON c.order_id = o.order_id

JOIN item_info i
ON o.item_id = i.item_id;

-- =====================================
-- Create Index
-- =====================================

CREATE INDEX customer_index
ON TABLE customer_info(cust_name)
AS 'org.apache.hadoop.hive.ql.index.compact.CompactIndexHandler'
WITH DEFERRED REBUILD;

-- =====================================
-- Rebuild Index
-- =====================================

ALTER INDEX customer_index
ON customer_info
REBUILD;

-- =====================================
-- Find Total Sales
-- =====================================

SELECT SUM(o.quantity * i.item_price) AS total_sales

FROM order_info o

JOIN item_info i
ON o.item_id = i.item_id;

-- =====================================
-- Find Average Sales
-- =====================================

SELECT AVG(o.quantity * i.item_price) AS average_sales

FROM order_info o

JOIN item_info i
ON o.item_id = i.item_id;

-- =====================================
-- Find Maximum Cost Order
-- =====================================

SELECT
c.cust_name,
o.order_id,
i.item_name,
o.quantity,
(i.item_price * o.quantity) AS total_cost

FROM customer_info c

JOIN order_info o
ON c.order_id = o.order_id

JOIN item_info i
ON o.item_id = i.item_id

ORDER BY total_cost DESC
LIMIT 1;

-- =====================================
-- EXTERNAL HIVE TABLE CONNECTED TO HBASE
-- =====================================

CREATE EXTERNAL TABLE customer_hbase (
    cust_id STRING,
    cust_name STRING,
    order_id STRING,
    item_name STRING,
    item_price STRING
)

STORED BY 'org.apache.hadoop.hive.hbase.HBaseStorageHandler'

WITH SERDEPROPERTIES (
"hbase.columns.mapping" =
":key,customer:name,order:orderid,item:itemname,item:price"
)

TBLPROPERTIES ("hbase.table.name" = "CustomerInfo");

-- =====================================
-- View HBase Data through Hive
-- =====================================
