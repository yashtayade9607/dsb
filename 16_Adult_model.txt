import pandas as pd
import numpy as np

from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from sklearn.preprocessing import StandardScaler

from sklearn.linear_model import LogisticRegression
from sklearn.naive_bayes import GaussianNB

from sklearn.metrics import accuracy_score

# -----------------------------------------------------
# Load Dataset
# -----------------------------------------------------

col_names = [
    "age",
    "workclass",
    "fnlwgt",
    "education",
    "education_num",
    "marital_status",
    "occupation",
    "relationship",
    "race",
    "sex",
    "capital_gain",
    "capital_loss",
    "hours_per_week",
    "native_country",
    "income"
]

df = pd.read_csv("adult.csv", names=col_names)

print("Original Dataset")
print(df.tail())

# -----------------------------------------------------
# m. Data Cleaning
# -----------------------------------------------------

# Replace ? with NaN
df = df.replace("?", np.nan)

# Remove missing values
df = df.dropna()

# Remove extra spaces
for col in df.columns:
    df[col] = df[col].astype(str).str.strip()

print("\nAfter Cleaning")
print(df.head())

# -----------------------------------------------------
# Remove Negative Values
# -----------------------------------------------------

numeric_cols = [
    "age",
    "fnlwgt",
    "education_num",
    "capital_gain",
    "capital_loss",
    "hours_per_week"
]

for col in numeric_cols:
    df = df[df[col].astype(int) >= 0]

# -----------------------------------------------------
# n. Error Correcting (Revised)
# -----------------------------------------------------
for col in numeric_cols:
    df[col] = df[col].astype(int)

    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1

    # Only filter if there is actual variance in the data
    if IQR > 0:
        lower = Q1 - 1.5 * IQR
        upper = Q3 + 1.5 * IQR
        
        # Count outliers before removing
        outlier_count = df[(df[col] < lower) | (df[col] > upper)].shape[0]
        df = df[(df[col] >= lower) & (df[col] <= upper)]
        print(f"Outliers removed in {col}: {outlier_count}")
    else:
        print(f"Skipping outliers for {col} (IQR is 0)")

print(f"\nFinal dataset shape after outlier removal: {df.shape}")

# -----------------------------------------------------
# o. Data Transformation
# Label Encoding
# -----------------------------------------------------

for col in df.columns:

    le = LabelEncoder()

    df[col] = le.fit_transform(df[col])

print("\nAfter Label Encoding")
print(df.head())

# -----------------------------------------------------
# Feature Scaling
# -----------------------------------------------------

X = df.drop("income", axis=1)

y = df["income"]

scaler = StandardScaler()

X_scaled = scaler.fit_transform(X)

# -----------------------------------------------------
# Split Dataset
# -----------------------------------------------------

X_train, X_test, y_train, y_test = train_test_split(
    X_scaled,
    y,
    test_size=0.2,
    random_state=42
)

# -----------------------------------------------------
# p. Logistic Regression
# -----------------------------------------------------

lr = LogisticRegression()

lr.fit(X_train, y_train)

y_pred_lr = lr.predict(X_test)

lr_accuracy = accuracy_score(y_test, y_pred_lr)

print("\nLogistic Regression Accuracy")
print(lr_accuracy)

# -----------------------------------------------------
# Gaussian Naive Bayes
# -----------------------------------------------------

nb = GaussianNB()

nb.fit(X_train, y_train)

y_pred_nb = nb.predict(X_test)

nb_accuracy = accuracy_score(y_test, y_pred_nb)

print("\nNaive Bayes Accuracy")
print(nb_accuracy)

# -----------------------------------------------------
# Accuracy Comparison
# -----------------------------------------------------

print("\nAccuracy Comparison")
print("Logistic Regression :", lr_accuracy)
print("Naive Bayes         :", nb_accuracy)
